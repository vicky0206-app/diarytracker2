    function setCurrentWeek() {

      const now = new Date();

      const weekStr = getISOWeekString(now);

      document.getElementById('weekly-week').value = weekStr;

      loadWeeklyData();

    function changeWeek(weeks) {

      const input = document.getElementById('weekly-week');

      if (!input.value) return setCurrentWeek();

      const [year, week] = input.value.split('-W').map(Number);

      

      const simpleDate = new Date(year, 0, 1 + (week - 1) * 7);

      simpleDate.setDate(simpleDate.getDate() + (weeks * 7));

      input.value = getISOWeekString(simpleDate);

      loadWeeklyData();

    }


    function getISOWeekString(date) {

      const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));

      const dayNum = d.getUTCDay() || 7;

      d.setUTCDate(d.getUTCDate() + 4 - dayNum);

      const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));

      const weekNo = Math.ceil((((d - yearStart) / 86400000) + 1) / 7);

      return `${d.getUTCFullYear()}-W${weekNo.toString().padStart(2, '0')}`;

    }

    function checkSelfcareStreakAlert(currentDateStr) {

      const banner = document.getElementById('selfcare-alert-banner');

      if (!banner || !currentDateStr) return;


      const currDate = new Date(currentDateStr);

      let streakCount = 0;


      for (let i = 0; i < 3; i++) {

        const d = new Date(currDate);

        d.setDate(d.getDate() - i);

        const dateStr = d.toISOString().split('T')[0];

        

        const entryStr = localStorage.getItem(`daily_${dateStr}`);

        if (entryStr) {

          const entry = JSON.parse(entryStr);

          if (entry.selfcare === 'gar_nicht' || entry.selfcare === 'wenig') {

            streakCount++;

          } else {

            break;

          }

        } else {

          break;

        }

      }


      if (streakCount >= 3) {

        banner.classList.remove('hidden');

        lucide.createIcons();

      } else {

        banner.classList.add('hidden');

      }

    }


    function checkCriticalAlerts(currentWeek, currentAvgScore) {

      const alertBanner = document.getElementById('critical-alert-banner');

      const alertReason = document.getElementById('alert-reason-text');


      let isCritical = false;

      let reasonMsg = "";


      if (currentAvgScore > 1.0) {

        isCritical = true;

        reasonMsg = `Dein aktueller Frühwarnzeichen-Score liegt bei ${currentAvgScore.toFixed(2)} (über dem Schwellenwert von 1.00).`;

      } 

      else {

        const allWeeklyKeys = Object.keys(localStorage)

          .filter(k => k.startsWith('weekly_'))

          .sort();


        if (allWeeklyKeys.length >= 4) {

          const scores = allWeeklyKeys.map(k => JSON.parse(localStorage.getItem(k)).avgScore);

          const len = scores.length;

          if (scores[len - 4] < scores[len - 3] &&

              scores[len - 2] < scores[len - 1]) {

            reasonMsg = "Dein Frühwarnzeichen-Score ist in den letzten 4 Wochen kontinuierlich angestiegen.";

        }

      }


      if (isCritical) {

        alertReason.innerText = reasonMsg;

        alertBanner.classList.remove('hidden');

      } else {

        alertBanner.classList.add('hidden');

      }

    }

    // Touch Swipe Steuerung (Wischen nach links/rechts)

    function initChartSwipeEvents() {

      const swipeArea = document.getElementById('charts-swipe-area');

      if (!swipeArea) return;


      let startX = 0;

      let startY = 0;

      let distX = 0;

      let distY = 0;


      swipeArea.addEventListener('touchstart', (e) => {

        const touch = e.touches[0];

        startX = touch.clientX;

        startY = touch.clientY;

        distX = 0;

        distY = 0;

      }, { passive: true });


      swipeArea.addEventListener('touchmove', (e) => {

        if (!startX || !startY) return;

        const touch = e.touches[0];

        distX = touch.clientX - startX;

        distY = touch.clientY - startY;

      }, { passive: true });


      swipeArea.addEventListener('touchend', () => {

        if (Math.abs(distX) > 40 && Math.abs(distX) > Math.abs(distY)) {

          if (distX > 0) {

            navigateChartOffset(1);

          } else {

            navigateChartOffset(-1);

          }

        }

        startX = 0;

        startY = 0;

      });

    }

    function selectHistoryDate(dateStr) {

      document.getElementById('daily-date').value = dateStr;

      switchTab('daily');

      loadDailyData();

    }


    function exportData() {

      const data = {};

      for (let i = 0; i < localStorage.length; i++) {

        const key = localStorage.key(i);

        data[key] = localStorage.getItem(key);

      }

      const jsonStr = JSON.stringify(data, null, 2);

      const blob = new Blob([jsonStr], { type: 'application/json' });

      const url = URL.createObjectURL(blob);


      const a = document.createElement('a');

      a.href = url;

      a.download = `diary_tracker_backup_${new Date().toISOString().split('T')[0]}.json`;

      a.click();

      URL.revokeObjectURL(url);

    }


    function importData(e) {

      const file = e.target.files[0];

      if (!file) return;


      const reader = new FileReader();

      reader.onload = function(event) {

        try {

          const imported = JSON.parse(event.target.result);

          Object.keys(imported).forEach(k => {

            localStorage.setItem(k, imported[k]);

          });

          showToast('Import erfolgreich abgeschlossen!');

          loadDailyData();

          loadWeeklyData();

          renderHistoryList();

          renderWeeklyCharts();

        } catch (err) {

          alert('Fehler beim Importieren der Datei. Bitte erstelle eine gültige JSON-Datei.');

        }

      };

      reader.readAsText(file);

    }


    function showToast(msg) {

      const toast = document.getElementById('toast');

      document.getElementById('toast-message').innerText = msg;

      toast.classList.remove('translate-y-20', 'opacity-0');

      setTimeout(() => {

        toast.classList.add('translate-y-20', 'opacity-0');

      }, 3000);

    }
